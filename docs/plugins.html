Plugin development

BetterPMMP keeps the plugin API of PocketMine-MP 5 (API 5.51.0). Every plugin written for PocketMine-MP 5 runs on BetterPMMP, and everything you learn here also works there. On top of that, BetterPMMP gives plugins access to game rules, scoreboards, target selectors and structures.

What you need

  • A BetterPMMP server for testing (see Installation).
  • An editor with PHP support, for example VS Code with the PHP Intelephense extension or PhpStorm.
  • The DevTools plugin, so the server loads plugins from folders while you develop.

Your first plugin

A plugin is a folder with a plugin.yml and PHP classes. Put it into the plugins folder of your server:

HelloWorld/
├── plugin.yml
└── src/
    └── Main.php

plugin.yml

name: HelloWorld
main: leondev\helloworld\Main
src-namespace-prefix: leondev\helloworld
version: 1.0.0
api: 5.0.0
author: YourName
description: My first BetterPMMP plugin

commands:
  hello:
    description: Says hello
    usage: "/hello [player]"
    permission: helloworld.command.hello

permissions:
  helloworld.command.hello:
    default: true

api: 5.0.0 means: works with every PocketMine-MP 5 API version, so also with BetterPMMP. With src-namespace-prefix the classes sit directly in src/.

src/Main.php

<?php

declare(strict_types=1);

namespace leondev\helloworld;

use pocketmine\command\Command;
use pocketmine\command\CommandSender;
use pocketmine\event\Listener;
use pocketmine\event\player\PlayerJoinEvent;
use pocketmine\plugin\PluginBase;
use pocketmine\utils\TextFormat;

final class Main extends PluginBase implements Listener{

	protected function onEnable() : void{
		$this->getServer()->getPluginManager()->registerEvents($this, $this);
		$this->getLogger()->info("Running on " . $this->getServer()->getName());
	}

	public function onJoin(PlayerJoinEvent $event) : void{
		$event->getPlayer()->sendMessage(TextFormat::GREEN . "Welcome to the server!");
	}

	public function onCommand(CommandSender $sender, Command $command, string $label, array $args) : bool{
		if($command->getName() === "hello"){
			$sender->sendMessage("Hello, " . ($args[0] ?? $sender->getName()) . "!");
			return true;
		}
		return false;
	}
}

Restart the server, join and type /hello. To share the plugin, build a .phar with DevTools (/makeplugin HelloWorld) and give others the file from plugin_data/DevTools.

Events

Mobs, items and blocks of BetterPMMP use the normal PocketMine classes, so the usual events work for them: damage, death, block break and place, interact, item use and so on.

use pocketmine\event\entity\EntityDamageByEntityEvent;
use pocketmine\event\entity\EntityDeathEvent;
use pocketmine\entity\Living;
use pocketmine\player\Player;

public function onDamage(EntityDamageByEntityEvent $event) : void{
	// BetterPMMP mobs are normal PocketMine entities
	if($event->getDamager() instanceof Living && $event->getEntity() instanceof Player){
		$event->setBaseDamage($event->getBaseDamage() * 0.5); // mobs hit half as hard
	}
}

public function onDeath(EntityDeathEvent $event) : void{
	$this->getLogger()->info($event->getEntity()->getName() . " died");
}

BetterPMMP APIs

These classes only exist on BetterPMMP. If your plugin should also run on plain PocketMine-MP, check first:

use pocketmine\VersionInfo;

$isBetterPMMP = $this->getServer()->getName() === "BetterPMMP";
if($isBetterPMMP){
	$this->getLogger()->info("BetterPMMP " . VersionInfo::BETTERPMMP_BUILD);
}

Game rules

pocketmine\world\GameRules reads and changes the 38 Bedrock game rules of a world, exactly like /gamerule.

use pocketmine\world\GameRules;

$world = $this->getServer()->getWorldManager()->getDefaultWorld();

// read (keys are the lowercase vanilla names)
if(GameRules::bool($world, "keepinventory")){
	// ...
}
$speed = GameRules::int($world, "randomtickspeed");

// write: saved in level.dat and sent to the players of the world
GameRules::set($world, "dodaylightcycle", false);
GameRules::set($world, "playerssleepingpercentage", 50);

Scoreboard

pocketmine\scoreboard\Scoreboard is the same scoreboard that /scoreboard uses. It is saved automatically and shown to all players. Display slots are sidebar, list and belowname.

use pocketmine\scoreboard\Scoreboard;

$board = Scoreboard::get();
if(!$board->hasObjective("kills")){
	$board->addObjective("kills", "dummy", "§cKills");
	$board->setDisplay("sidebar", "kills", 1); // 1 = highest first, 0 = lowest first
}

$name = $player->getName();
$board->setScore("kills", $name, ($board->getScore("kills", $name) ?? 0) + 1);

Target selectors

pocketmine\command\utils\TargetSelector resolves @a, @p, @e[type=zombie,r=10] and player names to entities.

use pocketmine\command\utils\TargetSelector;
use pocketmine\player\Player;

// in onCommand(): accepts a player name or a selector like @a[r=10]
foreach(TargetSelector::select($sender, $args[0] ?? "@s") as $entity){
	if($entity instanceof Player){
		$entity->sendMessage("You were selected!");
	}
}

Structures

pocketmine\world\structure\McStructure reads, writes and places .mcstructure files, the format of the structure block and of Bedrock add-ons.

use pocketmine\world\structure\McStructure;
use pocketmine\math\Vector3;

// load a .mcstructure file and place it, rotated by 90 degrees
$structure = McStructure::read(file_get_contents($this->getDataFolder() . "house.mcstructure"));
$structure->place($world, new Vector3(100, 64, 100), 1, "none", true, true, 1.0, 0);

// save an area as .mcstructure
$data = McStructure::capture($world, new Vector3(0, 60, 0), new Vector3(15, 80, 15), true, true)
	->write(new Vector3(0, 60, 0));
file_put_contents($this->getDataFolder() . "area.mcstructure", $data);

Plugins entwickeln

BetterPMMP behält die Plugin-API von PocketMine-MP 5 (API 5.51.0). Jedes Plugin für PocketMine-MP 5 läuft auf BetterPMMP, und alles, was du hier lernst, funktioniert auch dort. Zusätzlich gibt BetterPMMP Plugins Zugriff auf Spielregeln, Scoreboards, Zielauswahl und Strukturen.

Was du brauchst

  • Einen BetterPMMP-Server zum Testen (siehe Installation).
  • Einen Editor mit PHP-Unterstützung, zum Beispiel VS Code mit der Erweiterung PHP Intelephense oder PhpStorm.
  • Das Plugin DevTools, damit der Server beim Entwickeln Plugins aus Ordnern lädt.

Dein erstes Plugin

Ein Plugin ist ein Ordner mit einer plugin.yml und PHP-Klassen. Er kommt in den Ordner plugins deines Servers:

HelloWorld/
├── plugin.yml
└── src/
    └── Main.php

plugin.yml

name: HelloWorld
main: leondev\helloworld\Main
src-namespace-prefix: leondev\helloworld
version: 1.0.0
api: 5.0.0
author: YourName
description: My first BetterPMMP plugin

commands:
  hello:
    description: Says hello
    usage: "/hello [player]"

  helloworld.command.hello:

Events

Mobs, Items und Blöcke von BetterPMMP nutzen die normalen PocketMine-Klassen, deshalb funktionieren die üblichen Events auch für sie: Schaden, Tod, Blöcke abbauen und setzen, Interagieren, Items benutzen und so weiter.

use pocketmine\event\entity\EntityDamageByEntityEvent;
use pocketmine\event\entity\EntityDeathEvent;
use pocketmine\entity\Living;
use pocketmine\player\Player;

		$event->setBaseDamage($event->getBaseDamage() * 0.5); // mobs hit half as hard
	}
}

public function onDeath(EntityDeathEvent $event) : void{
